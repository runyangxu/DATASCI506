momentum_walk1 <- function(n, p) {
  if (length(n) != 1 || !is.numeric(n) || is.na(n) ||
      n < 1 || n != floor(n)) {
    stop("n must be a positive integer.")
  }
  
  if (length(p) != 1 || !is.numeric(p) || is.na(p) ||
      p < 0 || p > 1) {
    stop("p must be between 0 and 1.")
  }
  
  directions <- matrix(
    c(1, 0,
     -1, 0,
      0, 1,
      0, -1),
    ncol = 2,
    byrow = TRUE
  )
  
  previous <- sample(1:4, 1)
  position <- directions[previous, ]
  
  if (n > 1) {
    for (i in 2:n) {
      if (runif(1) < p) {
        current <- previous
      } else {
        current <- sample(1:4, 1)
      }
      
      position <- position + directions[current, ]
      previous <- current
    }
  }
  
  return(position)
}

momentum_walk1 uses a for loop to update the position one step at a time. The previous direction is stored so that it can be preserved with probability p at the next step.

Version 2

momentum_walk2 <- function(n, p) {
  if (length(n) != 1 || !is.numeric(n) || is.na(n) ||
      n < 1 || n != floor(n)) {
    stop("n must be a positive integer.")
  }
  
  if (length(p) != 1 || !is.numeric(p) || is.na(p) ||
      p < 0 || p > 1) {
    stop("p must be between 0 and 1.")
  }
  
  directions <- matrix(
    c(1, 0,
     -1, 0,
      0, 1,
      0, -1),
    ncol = 2,
    byrow = TRUE
  )
  
  previous <- sample(1:4, 1)
  position <- directions[previous, ]
  
  if (n > 1) {
    sapply(2:n, function(i) {
      if (runif(1) < p) {
        current <- previous
      } else {
        current <- sample(1:4, 1)
      }
      
      position <<- position + directions[current, ]
      previous <<- current
      
      return(NULL)
    })
  }
  
  return(position)
}

momentum_walk2(10, 0.6)
[1] 4 0
momentum_walk2(1000, 0.6)
[1] 52 10

momentum_walk2 uses sapply instead of an explicit loop. At each step, the function updates the current position and previous direction, while using the same momentum rule as the first implementation.
