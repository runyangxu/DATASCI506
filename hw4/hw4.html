aces <- tennis %>%

  summarise(
    mean_winner_aces = mean(w_ace),
    mean_loser_aces = mean(l_ace),
    mean_difference = mean(difference),
    percent_more_aces = 100 * mean(difference > 0)
  ) %>%
  kable(digits = 2)

7.5 5.79 1.7 57.28 
aces %>%
  ggplot(aes(x = difference)) +
  geom_histogram(binwidth = 2, fill = "steelblue", color = "white") +
  labs(title = "Difference in Aces (Winner - Loser)",
       x = "Ace difference", y = "Matches") +
  theme_minimal()

The distribution is roughly bell-shaped and centered near 0, with most matches having an ace difference between -10 and 10. This suggests that winners do not always have more aces than losers.

wins <- tennis %>%
  count(winner_id, winner_name, name = "wins") %>%
  rename(player_id = winner_id, player = winner_name)

losses <- tennis %>%
  count(loser_id, name = "losses") %>%
  rename(player_id = loser_id)

win_rates <- wins %>%
  full_join(losses, by = "player_id") %>%
  mutate(wins = replace_na(wins, 0L),
         losses = replace_na(losses, 0L),
         matches = wins + losses,
         win_rate = wins / matches) %>%
  filter(matches >= 5) %>%
  arrange(desc(win_rate))

win_rates %>%
  filter(win_rate == max(win_rate)) %>%
  select(player, wins, losses, matches, win_rate) %>%
  kable(digits = 3)
player wins losses matches win_rate

Problem 3 - Visualization

covid <- read_csv(
  "https://raw.githubusercontent.com/nytimes/covid-19-data/refs/heads/master/rolling-averages/us-states.csv",
  show_col_types = FALSE
) %>%
  filter(state %in% state.name) %>%
  mutate(date = as.Date(date))

1

us_cases <- covid %>%
  group_by(date) %>%
  summarise(cases_avg = sum(cases_avg, na.rm = TRUE), .groups = "drop")

ggplot(us_cases, aes(date, cases_avg)) +
  geom_line(color = "steelblue", linewidth = 0.5) +
  geom_smooth(se = FALSE, color = "firebrick", span = 0.04) +
  labs(title = "COVID-19 Waves in the United States",
       x = "Date", y = "Daily new cases (7-day average)") +
  scale_y_continuous(labels = scales::comma) +
  theme_minimal()
`geom_smooth()` using method = 'gam' and formula = 'y ~ s(x, bs = "cs")'

There are several major COVID-19 waves in the US. The largest peak occurred in early 2022, with nearly 800,000 daily new cases. The red smooth curve shows the overall trend more clearly.

2

state_rates <- covid %>%
  group_by(state) %>%
  summarise(total_rate = sum(cases_avg_per_100k, na.rm = TRUE),
            .groups = "drop") %>%
  arrange(desc(total_rate))

highest <- state_rates %>% slice_head(n = 1)
lowest <- state_rates %>% slice_tail(n = 1)
bind_rows(highest, lowest) %>% kable(digits = 1)
state total_rate
Rhode Island 43295.5
Maine 22020.2 
selected_states <- c(highest$state, lowest$state)

covid %>%
  filter(state %in% selected_states) %>%
  ggplot(aes(date, cases_avg_per_100k, color = state)) +
  geom_line(linewidth = 0.5) +
  facet_wrap(~state, ncol = 1) +
  labs(title = "Highest and Lowest Cumulative COVID-19 Case Rates",
       x = "Date", y = "Daily cases per 100,000 (7-day average)") +
  theme_minimal() +
  theme(legend.position = "none")

Rhode Island had much higher daily cases per 100,000 people than Maine, especially in early 2022. Rhode Island peaked at around 500, while Maine stayed below 100.
